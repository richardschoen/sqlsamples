# View TCP/IP Journal Entries
```
cl:CRTDUPOBJ OBJ(QATOFIPF) FROMLIB(QSYS) 
          OBJTYPE(*FILE) TOLIB(TMP) NEWOBJ(IPFILTER);

cl:DSPJRN JRN(QUSRSYS/QIPFILTER) 
       JRNCDE((M)) 
       ENTTYP((TF)) 
       OUTPUT(*OUTFILE) 
       OUTFILFMT(*TYPE4) 
       OUTFILE(TMP/IPFILTER) 
       ENTDTALEN(*VARLEN *CALC);
       
-- View the TCP journal entries
SELECT
    TFSRCA AS SOURCE_ADDRESS,
    TFSRCP AS SOURCE_PORT,
    TFDSTA AS DESTINATION_ADDRESS,
    TFDSTP AS DESTINATION_PORT
FROM TMP.IPFILTER;      
```
